# rakuda Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Hugo を置き換える Ruby gem `rakuda` を別リポジトリに作り、onigra.github.io ブログを URL 完全一致で移行する。

**Architecture:** ミニマル・パイプライン型。`LoadConfig → LoadContent → RenderPages → Emit → CopyStatic` を直列実行。ブログ repo は `site.yml` / `content/` / `layouts/` / `static/` のみ保持し、Gemfile は置かない。

**Tech Stack:** Ruby 3.3+, kramdown, kramdown-parser-gfm, front_matter_parser, rack, optparse, test/unit, ERB, Psych

## Global Constraints

- URL 互換: 全ページ完全一致（Post `/blog/YYYY/MM/DD/slug/`, Home `/` + `/page/N/`, Section `/post/`, Page `/about/`, Category `/categories/<slug>/`, RSS `/index.xml`, Robots `/robots.txt`）
- コンテンツ: 一度書き換えてよい（shortcode 削除、frontmatter 簡略化）
- SSG: 別 gem リポジトリ `rakuda`
- ブログ repo: Gemfile なし（CI で gem 参照）
- 開発: `serve` のみ（rebuild 手動、ライブリロードなし）
- テスト: test/unit（rspec 不可）
- CLI: optparse（thor 不可）
- runtime 依存: kramdown, kramdown-parser-gfm, front_matter_parser, rack のみ
- スコープ外: Disqus, GA, tags ページ, sitemap, gem migrate コマンド, ブログ Gemfile

```ruby
# lib/rakuda/url_generator.rb
module Rakuda
  class UrlGenerator
    def initialize(permalink_pattern)
      @pattern = permalink_pattern
    end

    def post_url(date, slug)
      @pattern
        .gsub(":year", date.strftime("%Y"))
        .gsub(":month", date.strftime("%m"))
        .gsub(":day", date.strftime("%d"))
        .gsub(":slug", slug)
    end

    def self.category_slug(name)
      name.downcase.gsub(/\s+/, "-")
    end

    def category_url(name)
      "/categories/#{self.class.category_slug(name)}/"
    end

    def home_page_url(page_num)
      page_num <= 1 ? "/" : "/page/#{page_num}/"
    end
  end
end
```

```ruby
# lib/rakuda/content_loader.rb
require "front_matter_parser"
require_relative "url_generator"

module Rakuda
  LoadResult = Data.define(:posts, :pages)

  class ContentLoader
    PARSER = FrontMatterParser::Parser.new(:md)

    def self.load(source_dir, config)
      new(source_dir, config).load
    end

    def initialize(source_dir, config)
      @source_dir = source_dir
      @config = config
      @url_gen = UrlGenerator.new(config.permalink_post)
    end

    def load
      posts = load_posts
      pages = load_pages
      LoadResult.new(posts: posts, pages: pages)
    end

    private

    def load_posts
      dir = File.join(@source_dir, "content", "post")
      return [] unless Dir.exist?(dir)

      Dir.glob(File.join(dir, "*.md")).map { |path| build_post(path) }
          .reject(&:draft)
          .sort_by(&:date).reverse
    end

    def load_pages
      path = File.join(@source_dir, "content", "about.md")
      return [] unless File.exist?(path)

      [build_page(path)]
    end

    def build_post(path)
      content = File.read(path)
      fm = PARSER.call(content)
      front = fm.front_matter
      body = fm.content
      date = Time.parse(front.fetch("date").to_s)
      slug = front.fetch("slug")
      summary, _rest = split_summary(body)

      Post.new(
        title: front.fetch("title"),
        slug: slug,
        date: date,
        categories: Array(front["categories"]),
        draft: front["draft"] == true,
        body: body,
        url: @url_gen.post_url(date, slug),
        summary: summary
      )
    end

    def build_page(path)
      content = File.read(path)
      fm = PARSER.call(content)
      front = fm.front_matter
      body = fm.content
      summary, = split_summary(body)

      Page.new(
        title: front.fetch("title"),
        url: front.fetch("url"),
        body: body,
        date: front["date"] ? Time.parse(front["date"].to_s) : nil,
        categories: Array(front["categories"]),
        summary: summary,
        content: nil
      )
    end

    def split_summary(body)
      parts = body.split("<!--more-->", 2)
      parts.length == 2 ? [parts[0].strip, parts[1].strip] : [body.strip, ""]
    end
  end
end
```

```ruby
module Rakuda
  PaginatorPage = Data.define(:items, :number, :total)

  module Paginator
    def self.paginate(items, per_page:)
      pages = items.each_slice(per_page).to_a
      total = pages.size
      pages.each_with_index.map do |slice, idx|
        PaginatorPage.new(items: slice, number: idx + 1, total: total)

    end
  end
```

```ruby
# lib/rakuda/template_renderer.rb
require "erb"

module Rakuda
  class TemplateRenderer
    def initialize(layouts_dir)
      @layouts_dir = layouts_dir
    end

    def render(template_name, assigns)
      path = File.join(@layouts_dir, "#{template_name}.erb")
      template = ERB.new(File.read(path))
      ctx = AssignContext.new(assigns)
      template.result(ctx.get_binding)
    end

    class AssignContext
      def initialize(assigns)
        assigns.each { |k, v| instance_variable_set("@#{k}", v) }
        assigns.each { |k, v| define_singleton_method(k) { v } }
      end

      def get_binding = binding
    end
  end
end
```

ERB テンプレート内では `site` / `page` メソッドとしてアクセス（`<%= page[:title] %>` 用に Hash をそのまま渡す）。

```ruby
# lib/rakuda/output_writer.rb
require "fileutils"

module Rakuda
  class OutputWriter
    def initialize(destination_dir)
      @dest = destination_dir
      FileUtils.mkdir_p(@dest)
    end

    def write_page(url, content)
      dir = url_to_dir(url)
      FileUtils.mkdir_p(dir)
      File.write(File.join(dir, "index.html"), content)
    end

    def write_file(relative_path, content)
      path = File.join(@dest, relative_path)
      FileUtils.mkdir_p(File.dirname(path))
      File.write(path, content)
    end

    def copy_static(source_dir)
      return unless Dir.exist?(source_dir)

      Dir.glob(File.join(source_dir, "**/*"), File::FNM_DOTMATCH).each do |src|
        next if File.directory?(src)
        rel = src.sub("#{source_dir}/", "")
        dest = File.join(@dest, rel)
        FileUtils.mkdir_p(File.dirname(dest))
        FileUtils.cp(src, dest)
      end
    end

    private

    def url_to_dir(url)
      path = url.sub(%r{/\z}, "")
      path.empty? ? @dest : File.join(@dest, path.sub(%r{\A/}, ""))
    end
  end
end
```

```ruby
# lib/rakuda/generators/post.rb
module Rakuda
  module Generators
    class Post
      def initialize(config:, posts:, renderer:, markdown:)
        @config = config
        @posts = posts
        @renderer = renderer
        @markdown = markdown
        @site = SiteContext.build(config)
      end

      def generate
        @posts.map do |post|
          content_html = @markdown.render(post.body)
          html = @renderer.render("single", {
            site: @site,
            page: post_to_page_hash(post, content_html)
          })
          { url: post.url, content: html }
        end
      end

      private

      def post_to_page_hash(post, content_html)
        {
          title: post.title,
          date: post.date,
          url: post.url,
          categories: post.categories,
          content: content_html,
          summary: @markdown.render(post.summary),
          is_home: false
        }
      end
    end
  end
end
```

```ruby
# lib/rakuda/pipeline.rb
module Rakuda
  class Pipeline
    def initialize(source:, destination:)
      @source = source
      @destination = destination
    end

    def run
      config = Config.load(File.join(@source, "site.yml"))
      loaded = ContentLoader.load(@source, config)
      writer = OutputWriter.new(@destination)
      renderer = TemplateRenderer.new(File.join(@source, "layouts"))
      markdown = MarkdownRenderer

      pages = []
      pages.concat Generators::Post.new(config:, posts: loaded.posts, renderer:, markdown:).generate
      pages.concat Generators::Home.new(config:, posts: loaded.posts, renderer:, markdown:).generate
      pages.concat Generators::Section.new(config:, posts: loaded.posts, renderer:, markdown:).generate
      pages.concat Generators::Taxonomy.new(config:, posts: loaded.posts, renderer:, markdown:).generate
      loaded.pages.each do |page|
        pages << Generators::Page.new(config:, page:, renderer:, markdown:).generate
      end

      pages.each { |p| writer.write_page(p[:url], p[:content]) }
      writer.write_file("index.xml", Generators::Rss.new(config:, posts: loaded.posts, renderer:).generate)
      writer.write_file("robots.txt", Generators::Robots.new(config:).generate)
      writer.copy_static(File.join(@source, "static"))
    end
  end
end
```

```ruby
# lib/rakuda/cli.rb
require "optparse"

module Rakuda
  class CLI
    def self.run(argv)
      new.run(argv)
    end

    def run(argv)
      command = argv.shift or abort usage
      options = { source: Dir.pwd, destination: "public", port: 4000 }

      OptionParser.new do |opts|
        opts.banner = "Usage: rkd #{command} [options]"
        opts.on("--source PATH") { |v| options[:source] = v }
        opts.on("--destination PATH") { |v| options[:destination] = v }
        opts.on("--port PORT", Integer) { |v| options[:port] = v }
      end.parse!(argv)

      case command
      when "build"
        Pipeline.new(source: options[:source], destination: options[:destination]).run
      when "serve"
        Server.start(root: options[:destination], port: options[:port])
      else
        abort usage
      end
    end

    def self.usage
      "Usage: rkd build|serve [--source PATH] [--destination public] [--port 4000]"
    end
  end
end
```
